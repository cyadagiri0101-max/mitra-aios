```http
POST /api/ekl/search
Content-Type: application/json
Authorization: Bearer <jwt-token>
x-tenant-id: 00000000-0000-0000-0000-000000000001

{
  "query": "What is the BM454 body insert material?",
  "topK": 5,
  "filters": {
    "projectNumber": "BM454",
    "entityType": "BOM_PART"
  },
  "minScore": 0.1
}
```
